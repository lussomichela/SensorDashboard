#PROJECT EXECUTION STEPS

#1. Prepare the environment 
#open two terminals and activate the virtual environment
source ~/zephyrproject/.venv/bin/activate
cd ~/zephyrproject/qt_bridge/Zephyr_qt_bridge

#2. Build the project
#terminal 1
rm -rf build
west build -b qemu_x86 .

#3. Start the dashboard (Qt application)

#4. Run QEMU
#terminal 1
/home/michela/zephyr-sdk-1.0.1/hosttools/sysroots/x86_64-pokysdk-linux/usr/bin/qemu-system-i386 \
  -m 128 \
  -nographic \
  -kernel build/zephyr/zephyr.elf \
  -serial unix:/tmp/qt_bridge.sock,server,nowait \
  -serial mon:stdio \
  -append "console=ttyS1"

#to switch between Zephyr logs and the QEMU monitor press "Ctrl + A" then "C"


#5. Run the bridge (bridge.py)
#terminal 2
python3 bridge.py
